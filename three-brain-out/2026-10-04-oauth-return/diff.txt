+// Google sign-in, returned through the website (test switch: /signup/?oauth=v2).
+//
+// The default path sends Google straight back to pursuit.honestecho.com, so the
+// website never sees the finished sign-in: the ad-click attribution never
+// reaches the user record and no conversion can be reported for a real account.
+// With the switch on, Google returns to /welcome on this origin instead. There
+// the session exists, so we can stamp attribution on a NEW account, report the
+// conversion, and then hand the session to pursuit the same way email signup
+// does (tokens in the URL hash).
+
+import type { Session } from '@supabase/supabase-js';
+import { supabase } from './supabase';
+import { track, getAttribution } from './analytics';
+import { reportSignupConversion } from './googleAds';
+
+const PENDING_KEY = 'he_oauth_pending';
+// A Google account that Supabase created this recently is a signup, not a
+// returning sign-in. Generous enough for a slow consent screen.
+const NEW_ACCOUNT_WINDOW_MS = 10 * 60 * 1000;
+
+export const OAUTH_RETURN_URL = 'https://honestecho.com/welcome';
+
+export function oauthReturnEnabled(): boolean {
+  return new URLSearchParams(window.location.search).get('oauth') === 'v2';
+}
+
+export function markOAuthPending(from: string | null): void {
+  try { sessionStorage.setItem(PENDING_KEY, JSON.stringify({ from })); } catch { /* flow falls back to the plain Welcome page */ }
+}
+
+export function clearOAuthPending(): void {
+  try { sessionStorage.removeItem(PENDING_KEY); } catch { /* nothing to clear */ }
+}
+
+function takeOAuthPending(): { from: string | null } | null {
+  try {
+    const raw = sessionStorage.getItem(PENDING_KEY);
+    if (!raw) return null;
+    sessionStorage.removeItem(PENDING_KEY);
+    return JSON.parse(raw) as { from: string | null };
+  } catch {
+    return null;
+  }
+}
+
+export function hasOAuthPending(): boolean {
+  try { return !!sessionStorage.getItem(PENDING_KEY); } catch { return false; }
+}
+
+function bridgeToPursuit(session: Session, isNewAccount: boolean): void {
+  // auth-js on pursuit rejects the hash unless expires_in AND token_type are
+  // present alongside the tokens (same contract as the email signup bridge).
+  const params = new URLSearchParams({
+    access_token: session.access_token,
+    refresh_token: session.refresh_token,
+    expires_in: String(session.expires_in ?? 3600),
+    ...(session.expires_at ? { expires_at: String(session.expires_at) } : {}),
+    token_type: session.token_type ?? 'bearer',
+    ...(isNewAccount ? { type: 'signup' } : {}),
+  });
+  window.location.replace(`https://pursuit.honestecho.com#${params.toString()}`);
+}
+
+// Runs on /welcome. Returns 'no-pending' when this visit is not an OAuth
+// return, 'no-session' when Google sign-in did not complete, and otherwise
+// navigates away to pursuit.
+type OAuthReturnResult = 'no-pending' | 'no-session' | 'bridged';
+let inFlight: Promise<OAuthReturnResult> | null = null;
+// Memoized: a double-mounted effect must not run the hand-off twice.
+export function finishOAuthReturn(): Promise<OAuthReturnResult> {
+  return (inFlight ??= runOAuthReturn());
+}
+
+async function runOAuthReturn(): Promise<OAuthReturnResult> {
+  const pending = takeOAuthPending();
+  if (!pending) return 'no-pending';
+
+  // getSession() waits for the client to finish reading the tokens Google's
+  // redirect left in the URL.
+  const { data } = await supabase.auth.getSession();
+  let session = data.session;
+  if (!session) return 'no-session';
+
+  const user = session.user;
+  const isNewAccount = Date.now() - new Date(user.created_at).getTime() < NEW_ACCOUNT_WINDOW_MS;
+
+  if (isNewAccount) {
+    const acquisition = { ...getAttribution(), ...(pending.from ? { from: pending.from } : {}) };
+    if (Object.keys(acquisition).length && !user.user_metadata?.acquisition) {
+      const { error } = await supabase.auth.updateUser({ data: { acquisition } });
+      if (error) console.warn('Could not save acquisition:', error.message);
+      // updateUser can rotate the session; bridge the current one.
+      session = (await supabase.auth.getSession()).data.session ?? session;
+    }
+    track('signup_account_created', { method: 'google', ...(pending.from ? { from: pending.from } : {}) });
+    await reportSignupConversion(user.id);
+  }
+
+  bridgeToPursuit(session, isNewAccount);
+  return 'bridged';
+}
